        public void SendAttendanceSheet()
        {
            string fileName = $"{DateTime.Now.ToString("yyyyMMddHHmmss")}_Attendance.xlsx";
            CreateExcelSheet(fileName);
            SendEmail(fileName);
        }

        private void CreateExcelSheet(string fileName, Dictionary<string, DateTime> attendanceData)
{
    // Create a new Excel package
    using (ExcelPackage excelPackage = new ExcelPackage())
    {
        // Add a new worksheet
        ExcelWorksheet worksheet = excelPackage.Workbook.Worksheets.Add("Attendance");

        // Set headers
        worksheet.Cells[1, 1].Value = "Name";
        worksheet.Cells[1, 2].Value = "Time";

        // Fill data
        int row = 2;
        foreach (var entry in attendanceData)
        {
            worksheet.Cells[row, 1].Value = entry.Key; // Name
            worksheet.Cells[row, 2].Value = entry.Value.ToString("yyyy-MM-dd HH:mm:ss"); // Time
            row++;
        }

        // Auto-fit columns
        worksheet.Cells.AutoFitColumns();

        // Save the Excel file
        FileInfo excelFile = new FileInfo(fileName);
        excelPackage.SaveAs(excelFile);
    }
}

                EnableSsl = true,
            };

            MailMessage mail = new MailMessage("from@example.com", "to@example.com")
            {
                Subject = "Attendance Sheet",
                Body = "Please find the attached attendance sheet.",
            };

            mail.Attachments.Add(new Attachment(fileName));

            client.Send(mail);
        }
